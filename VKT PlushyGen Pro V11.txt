// --- [FIXED LEGO PIECE: BULLETPROOF JSON PARSER] ---
const safeParseJSON = (text: string) => {
    try {
        const firstBracket = text.indexOf('[');
        const firstBrace = text.indexOf('{');
        let start = -1;
        if (firstBracket !== -1 && (firstBrace === -1 || firstBracket < firstBrace)) {
            start = firstBracket;
        } else {
            start = firstBrace;
        }
        const lastBracket = text.lastIndexOf(']');
        const lastBrace = text.lastIndexOf('}');
        let end = -1;
        if (lastBracket !== -1 && (lastBrace === -1 || lastBracket > lastBrace)) {
            end = lastBracket;
        } else {
            end = lastBrace;
        }
        if (start === -1 || end === -1 || end < start) return null;
        const jsonStr = text.substring(start, end + 1);
        return JSON.parse(jsonStr);
    } catch (e) {
        return null;
    }
};

// --- [STABLE LEGO PIECES] ---
const cropImage = (base64: string, box: [number, number, number, number]): Promise<string> => {
    return new Promise((resolve, reject) => {
        const img = new Image();
        img.onload = () => {
            const canvas = document.createElement('canvas');
            const ctx = canvas.getContext('2d');
            if (!ctx) return reject("Canvas error");
            const [ymin, xmin, ymax, xmax] = box;
            const w = img.width;
            const h = img.height;
            const left = (xmin / 1000) * w;
            const top = (ymin / 1000) * h;
            const width = ((xmax - xmin) / 1000) * w;
            const height = ((ymax - ymin) / 1000) * h;
            const padX = width * 0.15;
            const padY = height * 0.15;
            const finalX = Math.max(0, left - padX);
            const finalY = Math.max(0, top - padY);
            const finalW = Math.min(w - finalX, width + padX * 2);
            const finalH = Math.min(h - finalY, height + padY * 2);
            canvas.width = finalW; canvas.height = finalH;
            ctx.drawImage(img, finalX, finalY, finalW, finalH, 0, 0, finalW, finalH);
            resolve(canvas.toDataURL('image/png'));
        };
        img.onerror = () => reject("Image load error");
        img.src = `data:image/png;base64,${base64}`;
    });
};

const apiKey = ""; 

const fetchWithRetry = async (url: string, options: any, retries = 5): Promise<any> => {
    for (let i = 0; i < retries; i++) {
        try {
            const response = await fetch(url, options);
            const data = await response.json();
            if (response.ok) return data;
            await new Promise(r => setTimeout(r, Math.pow(2, i) * 1000));
        } catch (err) {
            if (i === retries - 1) throw err;
            await new Promise(r => setTimeout(r, Math.pow(2, i) * 1000));
        }
    }
};

const PROMPT_BASE = `**ROLE:** VKT PLUSHYGEN SUPREME ARCHITECT & BRANDING EXPERT (IQ 260).
**MISSION:** High-end production blueprint. 3 views (Front, Side, Back) in 1x3 row.
**SPATIAL LOGIC:** You are skilled in "Spatial Mapping". When a logo or text is added to a specific position (e.g., "Left Chest"), you must render it correctly across all 3 views based on 3D physics.
**BRANDING LOGIC:** - If instructed to "Change text", identify the old text on the character and replace it with the new content, keeping the same style.
- If a "Reference Image" is provided for a logo/accessory, extract the visual elements from it and place them according to the "Position" instruction.
**MATERIAL:** Premium minky fabric, micro-seams. Exactly 5 fingers.
**QUALITY:** 8K cinematic lighting, transparent background.`;

const generatePlushImage = async (characterBase64: string, batchInstructions: string, refImages: {label: string, data: string}[]) => {
    const parts: any[] = [
        { text: PROMPT_BASE + `\n**MODIFICATION BATCH:**\n${batchInstructions}\n**MODE:** Absolute Symmetry Master Pose.` },
        { inlineData: { mimeType: "image/png", data: characterBase64 } }
    ];
    refImages.forEach(img => {
        parts.push({ text: `Reference for "${img.label}" (This could be a logo source OR a position/context map):` });
        parts.push({ inlineData: { mimeType: "image/png", data: img.data } });
    });
    const payload = { contents: [{ parts }], generationConfig: { responseModalities: ['TEXT', 'IMAGE'] } };
    const result = await fetchWithRetry(`https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-image-preview:generateContent?key=${apiKey}`, { method: 'POST', body: JSON.stringify(payload) });
    const imagePart = result.candidates?.[0]?.content?.parts?.find((p: any) => p.inlineData);
    if (!imagePart) throw new Error("AI không thể lắp ráp thiết kế.");
    return `data:image/png;base64,${imagePart.inlineData.data}`;
};

const analyzeCharacterAndAccessories = async (base64: string) => {
    const prompt = `[IRON AUDIT] Detect branding and accessories.
    STRICT JSON ONLY: {"label": "name", "accessories": [{"id": "uniq1", "label": "Detail Name", "currentDescription": "color/style/text"}]}`;
    const payload = { contents: [{ parts: [{ text: prompt }, { inlineData: { mimeType: "image/png", data: base64 } }] }], generationConfig: { responseMimeType: "application/json" } };
    const result = await fetchWithRetry(`https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-09-2025:generateContent?key=${apiKey}`, { method: 'POST', body: JSON.stringify(payload) });
    const parsed = safeParseJSON(result.candidates[0].content.parts[0].text);
    return parsed || { label: "Character", accessories: [] };
};

const ImageModal = ({ imageUrl, onClose }: { imageUrl: string, onClose: () => void }) => {
    const [view, setView] = useState(0); 
    const [zoom, setZoom] = useState(1);
    const handleWheel = (e: React.WheelEvent) => setZoom(p => Math.min(Math.max(1, p + (e.deltaY > 0 ? -0.2 : 0.2)), 4));

    return (
        <div className="fixed inset-0 z-[110] bg-black/98 backdrop-blur-xl flex flex-col items-center justify-center p-4">
            <button onClick={onClose} className="absolute top-6 right-6 p-4 bg-white/10 rounded-full text-white hover:bg-white/20 transition-all"><X /></button>
            <div className="relative w-full h-full max-w-5xl overflow-hidden flex items-center justify-center" onWheel={handleWheel}>
                <div style={{ transform: `scale(${zoom})`, transition: 'transform 0.2s' }}>
                   <div className="w-[300%] flex" style={{ transform: `translateX(${-view * (100/3)}%)` }}>
                        {[0, 1, 2].map(i => <img key={i} src={imageUrl} className="w-1/3 h-auto max-h-[85vh] object-contain" alt="v" />)}
                   </div>
                </div>
            </div>
            <div className="absolute bottom-8 bg-slate-900/90 border border-slate-700 p-2 rounded-2xl flex gap-2 backdrop-blur-md">
                {['Mặt trước', 'Mặt nghiêng', 'Mặt sau'].map((t, i) => (
                    <button key={i} onClick={() => { setView(i); setZoom(1); }} className={`px-6 py-2 rounded-xl text-xs font-black transition-all ${view === i ? 'bg-amber-500 text-slate-950 shadow-lg' : 'text-white'}`}>{t}</button>
                ))}
                <div className="w-px bg-slate-800 mx-2" />
                <button onClick={() => setZoom(1)} className="p-2 text-white hover:bg-slate-800 rounded-lg"><Maximize size={18}/></button>
            </div>
        </div>
    );
};

    // Branding & Custom Addition States

    const [newAccDesc, setNewAccDesc] = useState("");
    const [newAccPos, setNewAccPos] = useState(""); // Vị trí mới
    
    const [editPrompt, setEditPrompt] = useState("");

    const [isApplying, setIsApplying] = useState(false);

    const handleUpload = async (file: File) => {
        setUploadedFile(file);
        setState(AppState.LOADING);
        setError(null);
        try {
            const base64 = await new Promise<string>((r) => {
                const reader = new FileReader();
                reader.readAsDataURL(file);
                reader.onload = () => r((reader.result as string).split(',')[1]);
            });

            const detectRes = await fetchWithRetry(`https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-09-2025:generateContent?key=${apiKey}`, { 
                method: 'POST',

                    contents: [{ parts: [{ text: "DETECT ALL CHARACTERS. JSON ONLY." }, { inlineData: { mimeType: "image/png", data: base64 } }] }], 
                    generationConfig: { responseMimeType: "application/json" } 
                })

            const rawText = detectRes.candidates[0].content.parts[0].text;
            const parsedDetects = safeParseJSON(rawText);
            const detects = Array.isArray(parsedDetects) ? parsedDetects : (parsedDetects?.box_2d ? [parsedDetects] : []);
            
            if (detects.length === 0) throw new Error("Không phát hiện được thực thể.");

            const processed = await Promise.all(detects.map(async (d: any, i: number) => {
                const cropped = await cropImage(base64, d.box_2d);
                const analysis = await analyzeCharacterAndAccessories(cropped.split(',')[1]);
                return { 
                    id: `char-${i}`, image: cropped, label: analysis.label || d.label || "Nhân vật", box_2d: d.box_2d, 
                    accessories: Array.isArray(analysis.accessories) ? analysis.accessories : []

            }));

            setError(err.message);
            setState(AppState.ERROR);
        }
    };

    const handleInitialGenerate = async (char: DetectedCharacter) => {
        setState(AppState.LOADING);
        setActiveChar(char);
        try {
            const res = await generatePlushImage(char.image.split(',')[1], "Initial plush design", []);
            setDesign(res);
            setState(AppState.RESULT);
        } catch (err: any) {
            setError(err.message);
            setState(AppState.ERROR);
        }
    };

    const stageMutation = (type: 'modify' | 'remove') => {
        if (!selectedAcc) return;
        const instruction = type === 'remove' ? `REMOVE ${selectedAcc.label} completely.` : `BRANDING/STYLE MOD: ${editPrompt}. Color: ${editColor}.`;
        setStagedMutations(prev => ({ ...prev, [selectedAcc.id]: { id: selectedAcc.id, label: selectedAcc.label, type, instruction, color: editColor } }));
        setSelectedAcc(null);
        setEditPrompt("");
    };

    const stageBrandingAddition = () => {
        const id = `branding-${Date.now()}`;
        const instruction = newAccRefImage 
            ? `BRANDING/LOGO INJECTION: Extract the branding/logo from the reference image and place it exactly at [${newAccPos}]. Description: ${newAccDesc}`
            : `ADD NEW ELEMENT "${newAccLabel}" at [${newAccPos}]: ${newAccDesc}`;
            
        setStagedMutations(prev => ({
            ...prev,
            [id]: { id, label: newAccLabel || "Branding", type: 'branding', instruction, position: newAccPos, refImage: newAccRefImage || undefined }
        }));
        
        setIsAddingNew(false); setNewAccLabel(""); setNewAccRefImage(null); setNewAccDesc(""); setNewAccPos("");
    };

    const applyBatchMutations = async () => {

            `}</style>
            <div className="absolute inset-0 bg-grid opacity-10 pointer-events-none" />
            
            <Header />

                {state === AppState.IDLE && (
                    <div className="w-full max-w-3xl mx-auto bg-slate-900/40 backdrop-blur-2xl rounded-[3rem] p-8 sm:p-12 border border-slate-800 shadow-2xl mt-8">

                                    {selectedAcc && (
                                        <motion.div initial={{ opacity: 0, height: 0 }} animate={{ opacity: 1, height: 'auto' }} exit={{ opacity: 0, height: 0 }} className="mt-6 pt-6 border-t border-slate-800 space-y-4">
                                            <div className="flex items-center justify-between"><h4 className="font-black text-[10px] text-amber-500 uppercase tracking-widest flex items-center gap-2"><Settings2 size={12}/> {selectedAcc.label}</h4><button onClick={() => setSelectedAcc(null)} className="text-slate-500 p-1"><X size={14}/></button></div>
                                            <div className="space-y-4">
                                                <div className="bg-slate-950 p-3 rounded-xl border border-slate-800">
                                                    <p className="text-[9px] font-black text-slate-600 mb-2 uppercase">Chỉnh màu</p>
                                                    <div className="flex items-center gap-3"><input type="color" value={editColor} onChange={(e) => setEditColor(e.target.value)} className="w-10 h-8 rounded bg-slate-900 border-none cursor-pointer" /><span className="text-[10px] font-mono text-slate-400">{editColor}</span></div>
                                                </div>
                                                <textarea value={editPrompt} onChange={(e) => setEditPrompt(e.target.value)} placeholder="Thay đổi nội dung Branding/Text..." className="w-full bg-slate-950 border border-slate-800 rounded-xl p-3 text-xs focus:border-amber-500 outline-none h-16 text-white" />
                                                <div className="flex gap-2">
                                                    <button onClick={() => stageMutation('remove')} className="flex-1 py-3 bg-red-500/10 border border-red-500/20 text-red-500 rounded-xl font-black text-[9px] uppercase">Xóa</button>
                                                    <button onClick={() => stageMutation('modify')} className="flex-1 py-3 bg-amber-500 text-slate-950 rounded-xl font-black text-[9px] uppercase">Lưu</button>
                                                </div>
                                            </div>
                                        </motion.div>
                                    )}
                                </AnimatePresence>

                                <div className="mt-8 space-y-3">
                                    {Object.keys(stagedMutations).length > 0 && (
                                        <button onClick={applyBatchMutations} disabled={isApplying} className="w-full py-5 bg-gradient-to-r from-amber-600 to-amber-500 text-slate-950 rounded-2xl font-black text-sm uppercase flex items-center justify-center gap-3 shadow-2xl hover:brightness-110 disabled:opacity-50">
                                            {isApplying ? <RefreshCcw size={20} className="animate-spin" /> : <Zap size={20} fill="currentColor" />} Áp dụng {Object.keys(stagedMutations).length} Lego
                                        </button>
                                    )}
                                    <button onClick={() => { setState(AppState.IDLE); setStagedMutations({}); }} className="w-full py-3 border border-slate-800 text-slate-500 rounded-2xl font-bold text-[9px] uppercase flex items-center justify-center gap-2 hover:text-white"><RefreshCcw size={12} /> Hủy dự án</button>
                                </div>
                            </div>
                        </div>

                        {/* VIEWPORT */}
                        <div className="lg:col-span-8 flex flex-col gap-6">
                            <div className="bg-slate-900/60 p-8 sm:p-12 rounded-[3.5rem] border border-slate-800 relative shadow-2xl backdrop-blur-3xl flex flex-col min-h-[650px]">
                                {isApplying && (
                                    <div className="absolute inset-0 z-50 bg-slate-950/80 backdrop-blur-xl flex flex-col items-center justify-center text-center">
                                        <div className="w-14 h-14 border-4 border-amber-500 border-t-transparent rounded-full animate-spin mb-6" />
                                        <h3 className="text-amber-500 font-black text-lg tracking-widest uppercase animate-pulse">Ánh xạ Không gian & Nhúng Logo...</h3>
                                    </div>
                                )}
                                <div className="flex flex-col sm:flex-row justify-between items-start sm:items-center gap-6 mb-8">
                                    <div><h2 className="text-3xl font-black text-white tracking-tighter uppercase leading-none">Blueprint</h2><p className="text-amber-500/80 font-bold text-[8px] uppercase tracking-[0.4em] mt-3">Production High-End • V11.0 Spatial</p></div>
                                    <button onClick={() => { const a = document.createElement('a'); a.href = design; a.download = 'vkt-branding.png'; a.click(); }} className="px-6 py-4 bg-amber-500 text-slate-950 rounded-2xl font-black text-[10px] uppercase shadow-xl hover:bg-amber-400 flex items-center gap-2"><Download size={16} /> Tải bản vẽ</button>
                                </div>
                                <div className="bg-slate-950 rounded-[3rem] border border-slate-800 p-8 group relative overflow-hidden shadow-inner flex-grow flex items-center justify-center cursor-pointer" onClick={() => setShowModal(true)}>
                                    <img src={design} className="w-full h-auto relative z-10" alt="Design Result" />
                                    <div className="absolute inset-0 bg-black/40 opacity-0 group-hover:opacity-100 flex items-center justify-center transition-all z-20 backdrop-blur-sm">
                                        <div className="px-8 py-3 bg-white text-slate-950 font-black rounded-full text-[10px] uppercase tracking-widest shadow-2xl">Phóng lớn thiết kế</div>
                                    </div>
                                </div>
                                
                                <div className="mt-8 p-6 bg-slate-950/50 border border-slate-800 rounded-[2rem] flex items-start gap-4">
                                    <div className="p-3 bg-amber-500/10 rounded-full text-amber-500 shrink-0"><Target size={18}/></div>
                                    <div className="text-left">
                                        <p className="text-slate-200 text-[10px] font-black uppercase tracking-widest mb-1">Mảnh ghép Định danh (Spatial Intelligence)</p>
                                        <p className="text-slate-500 text-[10px] leading-relaxed font-medium">
                                            Để nhúng Logo chính xác: Nhấn [+] {'→'} Tải ảnh Logo {'→'} Nhập vị trí (VD: Ngực trái). Nếu bạn chụp màn hình một vị trí cụ thể và tải lên, hệ thống sẽ tự động ánh xạ Logo vào đúng "tọa độ" đó trên bản vẽ kỹ thuật.
                                        </p>
                                    </div>
                                </div>
                            </div>
                        </div>
                    </div>
                )}

                {state === AppState.ERROR && error && (
                    <div className="bg-slate-900/40 p-12 rounded-[4rem] border border-red-500/20 text-center max-w-lg mx-auto shadow-2xl backdrop-blur-3xl mt-20">
                        <AlertTriangle className="w-14 h-14 text-red-500 mx-auto mb-6" />
                        <h2 className="text-xl font-black text-white mb-4 uppercase">System Error</h2>
                        <p className="text-slate-400 mb-8 text-xs italic">{error}</p>
                        <button onClick={() => setState(AppState.IDLE)} className="w-full py-4 bg-slate-800 text-white rounded-2xl font-black uppercase text-[10px]">Thử lại</button>
                    </div>
                )}
            </main>
            {showModal && design && <ImageModal imageUrl={design} onClose={() => setShowModal(false)} />}
        </div>
    );
}
